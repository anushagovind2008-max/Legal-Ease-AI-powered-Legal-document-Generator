def render_template(template_name: str, values: dict) -> str:
    """Safely renders template with user provided inputs."""
    tpl = TEMPLATES[template_name]["template"]
    try:
        return tpl.format(**values)
    except KeyError:
        out = tpl
        for k, v in values.items():
            out = out.replace("{" + k + "}", str(v))
        return out

def generate_legal_pdf(document_text: str, title: str = "LEGAL AGREEMENT") -> bytes:
    """Generates professional legal PDF using ReportLab."""
    try:
        from reportlab.lib.pagesizes import letter
        from reportlab.lib import colors
        from reportlab.lib.styles import getSampleStyleSheet, ParagraphStyle
        from reportlab.platypus import SimpleDocTemplate, Paragraph, Spacer
        from reportlab.pdfgen import canvas

        buffer = io.BytesIO()
        doc = SimpleDocTemplate(buffer, pagesize=letter, leftMargin=54, rightMargin=54, topMargin=54, bottomMargin=54)
        styles = getSampleStyleSheet()

        title_style = ParagraphStyle(
            'DocTitle', parent=styles['Heading1'], fontName='Helvetica-Bold', fontSize=14, leading=17,
            textColor=colors.HexColor("#0F172A"), alignment=1, spaceAfter=14
        )
        heading_style = ParagraphStyle(
            'SectionHeader', parent=styles['Heading2'], fontName='Helvetica-Bold', fontSize=10, leading=13,
            textColor=colors.HexColor("#1E3A8A"), spaceBefore=8, spaceAfter=4
        )
        body_style = ParagraphStyle(
            'DocBody', parent=styles['Normal'], fontName='Helvetica', fontSize=9, leading=13,
            textColor=colors.HexColor("#1E293B"), spaceAfter=5
        )

        story = [Paragraph(title.upper(), title_style), Spacer(1, 8)]

        for line in document_text.split("\n"):
            stripped = line.strip()
            if not stripped:
                story.append(Spacer(1, 4))
                continue
            clean = stripped.replace("&", "&amp;").replace("<", "&lt;").replace(">", "&gt;")
            if stripped.startswith("###") or (len(stripped) > 2 and stripped[:2].isdigit() and stripped[2] == ".") or stripped.isupper():
                story.append(Paragraph(clean.lstrip("#").strip(), heading_style))
            else:
                story.append(Paragraph(clean, body_style))

        doc.build(story)
        buffer.seek(0)
        return buffer.getvalue()
    except Exception as e:
        # Fallback to plain text bytes if reportlab has issues
        return document_text.encode("utf-8")


def generate_legal_docx(document_text: str, title: str = "LEGAL AGREEMENT") -> bytes:
    """Generates styled Word (.docx) document using python-docx."""
    try:
        from docx import Document
        from docx.shared import Pt, Inches, RGBColor
        from docx.enum.text import WD_ALIGN_PARAGRAPH

        doc = Document()
        for section in doc.sections:
            section.top_margin = Inches(1.0)
            section.bottom_margin = Inches(1.0)
            section.left_margin = Inches(1.0)
            section.right_margin = Inches(1.0)

        # Title
        p_title = doc.add_paragraph()
        p_title.alignment = WD_ALIGN_PARAGRAPH.CENTER
        run = p_title.add_run(title.upper())
        run.bold = True
        run.font.name = "Arial"
        run.font.size = Pt(15)
        run.font.color.rgb = RGBColor(30, 58, 138)
        p_title.paragraph_format.space_after = Pt(14)

        for line in document_text.split("\n"):
            stripped = line.strip()
            if not stripped:
                continue
            p = doc.add_paragraph()
            p.paragraph_format.line_spacing = 1.15
            p.paragraph_format.space_after = Pt(5)

            if stripped.startswith("###") or (len(stripped) > 2 and stripped[:2].isdigit() and stripped[2] == ".") or stripped.isupper():
                clean_heading = stripped.lstrip("#").strip()
                r = p.add_run(clean_heading)
                r.bold = True
                r.font.name = "Arial"
                r.font.size = Pt(11)
                r.font.color.rgb = RGBColor(30, 58, 138)
                p.paragraph_format.space_before = Pt(8)
            else:
                r = p.add_run(stripped)
                r.font.name = "Arial"
                r.font.size = Pt(10)
                r.font.color.rgb = RGBColor(35, 35, 35)

        buf = io.BytesIO()
        doc.save(buf)
        buf.seek(0)
        return buf.getvalue()
    except Exception:
        return document_text.encode("utf-8")

def call_gemini_ai(prompt: str, api_key: str = None) -> str:
    """Calls Gemini via google.genai or legacy SDK with fallback."""
    key = api_key or os.getenv("GEMINI_API_KEY", "").strip()
    if not key:
        return None

    # Try modern google.genai SDK
    try:
        from google import genai
        client = genai.Client(api_key=key)
        res = client.models.generate_content(model="gemini-2.5-flash", contents=prompt)
        if res and res.text:
            return res.text.strip()
    except Exception:
        pass

    # Try legacy SDK
    try:
        import google.generativeai as legacy_genai
        legacy_genai.configure(api_key=key)
        model = legacy_genai.GenerativeModel("gemini-1.5-flash")
        res = model.generate_content(prompt)
        if res and res.text:
            return res.text.strip()
    except Exception:
        pass

    return None


def generate_ai_contract(prompt: str, doc_type: str, jurisdiction: str, clauses: list, api_key: str = None):
    clauses_str = ", ".join(clauses) if clauses else "Standard boilerplate clauses"
    full_prompt = f"""
You are an expert senior legal counsel specializing in corporate and commercial law.
Draft an exhaustive, professionally structured, legally enforceable agreement.
Do NOT use markdown code fences (```); output raw clean legal document text.

Document Type: {doc_type}
Governing Jurisdiction: {jurisdiction}
Mandatory Clauses to Include: {clauses_str}
Client Requirements / Terms:
{prompt}

Draft the complete legal agreement now:
"""
    ai_text = call_gemini_ai(full_prompt, api_key=api_key)
    if ai_text:
        return ai_text, True

    # Offline Smart Synthesizer
    fallback = f"""CUSTOM LEGAL AGREEMENT: {doc_type.upper()}

This Agreement is made effective as of the date of execution, governed by the laws of {jurisdiction}.

1. PARTIES AND TRANSACTIONAL SCOPE
The Parties formalize their binding obligations concerning the following specifications:
{prompt}

2. OPERATIONAL TERMS AND MANDATORY COVENANTS
"""
    for idx, c in enumerate(clauses, 1):
        fallback += f"\n2.{idx} {c.upper()}\nThe Parties agree to incorporate standard legal provisions for {c.lower()} pursuant to the prevailing statutory codes of {jurisdiction}.\n"

    fallback += f"""
3. GOVERNING LAW AND JURISDICTION
This Agreement shall be governed by, and construed in accordance with, the laws of {jurisdiction}. Any dispute arising hereunder shall be submitted exclusively to the competent judicial courts of {jurisdiction}.

4. SEVERABILITY AND INTEGRATION
If any term is held invalid, remaining provisions continue in full effect. This instrument represents the complete understanding between Parties.

IN WITNESS WHEREOF, the Parties execute this Agreement:

------------------------------------------------------
PARTY A: _________________________    Date: _______________
PARTY B: _________________________    Date: _______________
"""
    return fallback, False


def audit_contract_risks(text: str, api_key: str = None) -> str:
    prompt = f"You are a contract audit specialist. Audit this legal contract for risk level (Low/Medium/High), missing protections, loopholes, and recommendations:\n\n{text[:8000]}"
    res = call_gemini_ai(prompt, api_key)
    if res:
        return res

    # Heuristic audit
    tl = text.lower()
    findings = []
    if "indemnif" not in tl:
        findings.append("⚠️ **Missing Indemnification**: Neither party is explicitly protected against third-party liability.")
    else:
        findings.append("✅ **Indemnification Present**: Mutual or unilateral liability protections detected.")

    if "governing law" not in tl and "jurisdiction" not in tl:
        findings.append("⚠️ **Ambiguous Jurisdiction**: No clear forum identified for dispute resolution.")
    else:
        findings.append("✅ **Governing Jurisdiction Stated**: Clear legal forum identified.")

    if "severability" not in tl:
        findings.append("ℹ️ **No Severability Clause**: An invalid clause may jeopardize the whole agreement.")
    else:
        findings.append("✅ **Severability Clause Included**: Protects the validity of remaining provisions.")

    if "terminate" not in tl and "termination" not in tl:
        findings.append("⚠️ **Missing Exit Mechanism**: Unclear how parties terminate the agreement.")
    else:
        findings.append("✅ **Termination Terms Defined**: Formal exit mechanisms are present.")

    return "### 🛡️ Contract Risk & Vulnerability Audit\n\n" + "\n\n".join(findings) + "\n\n*(Connect Gemini API Key for deep AI auditing)*"


def explain_in_plain_english(text: str, api_key: str = None) -> str:
    prompt = f"Explain this legal contract in plain English for a non-lawyer with Executive Summary, Obligations, Deadlines/Money, Termination, and Gotchas:\n\n{text[:8000]}"
    res = call_gemini_ai(prompt, api_key)
    if res:
        return res

    return """### 📋 Plain English Summary (Smart Rule-Based Analysis)
- **Primary Objective**: Formal contract establishing rights, obligations, and deliverables between parties.
- **Key Obligations**: Ensure performance of stated services, adherence to confidentiality, and compliance with payment schedules.
- **Termination**: Contract provides terms for both termination with notice or immediate termination upon material breach.
- **Governing Law**: Governed by the specified jurisdiction's statutes.
- **Tip**: *Connect a Gemini API key in the sidebar for in-depth, clause-by-clause AI semantic translation.*"""


def refine_document_with_ai(original: str, instruction: str, api_key: str = None):
    prompt = f"Update and rewrite this legal document based precisely on this instruction:\nINSTRUCTION: {instruction}\n\nORIGINAL:\n{original}\n\nRETURN THE UPDATED FULL DOCUMENT:"
    res = call_gemini_ai(prompt, api_key)
    if res:
        return res, True
    amended = f"{original}\n\n[AMENDMENT / ADDENDUM]\nPursuant to instruction: {instruction}\nThe parties adopt these modifications effective immediately.\n"
    return amended, False
